  \item
  \if\relax\detokenize{#2}\relax
    #1\par
  \else
    \settowidth{\resumeRightWidth}{#2}
    \ifdim\resumeRightWidth>0.35\linewidth
      \setlength{\resumeRightWidth}{0.35\linewidth}
    \fi
    \noindent\begin{minipage}[t]{\dimexpr\linewidth-\resumeRightWidth-1em\relax}
      \raggedright\resumeBodyFont #1\par
    \end{minipage}\hfill
    \begin{minipage}[t]{\resumeRightWidth}
      \raggedleft\resumeBodyFont #2\par
    \end{minipage}\par
  \fi
  \nobreak
}
